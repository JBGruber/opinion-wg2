Function to colour-code ICR results

# 𝛼-levels:
#
# 𝛼 > 0.8 reliable
# 𝛼 > 0.667 and 𝛼 < 0.8 tentative conclusions
# 𝛼 < 0.667 not reliable
# (Krippendorff, 2004, pp. 241-243)
icr_print <- function(
  tbl,
  cols = c("Agreement", "Cohens_Kappa", "Krippendorffs_Alpha")
) {
  alpha_color <- function(x) {
    seg_high <- scales::col_numeric(c("#90EE90", "#006400"), domain = c(0.8, 1)) # light → dark green
    seg_mid <- scales::col_numeric(
      c("#FFA500", "#ADFF2F"),
      domain = c(0.667, 0.8)
    )
    seg_low <- scales::col_numeric(
      c("#8B0000", "#FF6B6B"),
      domain = c(-1, 0.667)
    )

    out <- character(length(x))
    high <- !is.na(x) & x >= 0.8
    mid <- !is.na(x) & x >= 0.667 & x < 0.8
    low <- !is.na(x) & x < 0.667
    out[high] <- seg_high(x[high])
    out[mid] <- seg_mid(x[mid])
    out[low] <- seg_low(x[low])
    out[is.na(x)] <- "#FFFFFF"
    out
  }

  tbl |>
    gt() |>
    fmt_number(
      columns = all_of(cols),
      decimals = 2L,
      drop_trailing_zeros = TRUE
    ) |>
    data_color(
      columns = all_of(cols),
      fn = alpha_color
    )
}

Machine learning metrics

Alternatively to the ICR metrics, we can also calculate more classic machine learning metrics: precision, recall and F1 scores. Since we have two annotators, I calculate these measures for both of them separately:

annotation_df_wide <- annotation_df |>
  pivot_longer(
    cols = all_of(compare_cols),
    names_to = "variable",
    values_to = "annotation"
  ) |>
  pivot_wider(
    id_cols = tool,
    names_from = c(annotator, variable),
    values_from = annotation
  ) |>
  mutate(across(contains("_category"), function(x) {
    factor(
      x,
      levels = c(
        "algorithm",
        "approach",
        "commercial_api",
        "data_collection",
        "general_library",
        "general_purpose_nlp",
        "gui_tool",
        "infrastructure",
        "linguistic_resource",
        "model",
        "other_unclear",
        "programming_language",
        "single_purpose_nlp"
      )
    )
  }))
