Function to colour-code ICR results

# 𝛼-levels:
#
# 𝛼 > 0.8 reliable
# 𝛼 > 0.667 and 𝛼 < 0.8 tentative conclusions
# 𝛼 < 0.667 not reliable
# (Krippendorff, 2004, pp. 241-243)
icr_print <- function(
  tbl,
  cols = c("Agreement", "Krippendorffs_Alpha")
) {
  alpha_color <- function(x) {
    seg_high <- scales::col_numeric(c("#90EE90", "#006400"), domain = c(0.8, 1)) # light → dark green
    seg_mid <- scales::col_numeric(
      c("#FFA500", "#ADFF2F"),
      domain = c(0.667, 0.8)
    )
    seg_low <- scales::col_numeric(
      c("#8B0000", "#FF6B6B"),
      domain = c(-1, 0.667)
    )

    out <- character(length(x))
    high <- !is.na(x) & x >= 0.8
    mid <- !is.na(x) & x >= 0.667 & x < 0.8
    low <- !is.na(x) & x < 0.667
    out[high] <- seg_high(x[high])
    out[mid] <- seg_mid(x[mid])
    out[low] <- seg_low(x[low])
    out[is.na(x)] <- "#FFFFFF"
    out
  }

  tbl |>
    gt() |>
    fmt_number(
      columns = all_of(cols),
      decimals = 2L,
      drop_trailing_zeros = TRUE
    ) |>
    data_color(
      columns = all_of(cols),
      fn = alpha_color
    )
}

Majority vs the machine

# row-wise majority vote across the three human coders.
# with 3 coders a binary (yes/no) variable always has a majority; for the
# multi-class category a tie (all three disagree) returns NA and is dropped
# from the metrics below.
majority_vote <- function(...) {
  vals <- as.character(c(...))
  vals <- vals[!is.na(vals)]
  if (length(vals) == 0) {
    return(NA_character_)
  }
  tab <- sort(table(vals), decreasing = TRUE)
  if (length(tab) > 1 && tab[[1]] == tab[[2]]) {
    return(NA_character_)
  }
  names(tab)[1]
}

annotation_df_wide_maj <- annotation_df_wide |>
  rowwise() |>
  mutate(
    MAJ_category = majority_vote(DT_category, CV_category, JG_category),
    MAJ_task_representation = majority_vote(
      DT_task_representation,
      CV_task_representation,
      JG_task_representation
    ),
    MAJ_task_classification = majority_vote(
      DT_task_classification,
      CV_task_classification,
      JG_task_classification
    ),
    MAJ_task_target = majority_vote(
      DT_task_target,
      CV_task_target,
      JG_task_target
    ),
    MAJ_software = majority_vote(DT_software, CV_software, JG_software)
  ) |>
  ungroup() |>
  mutate(
    MAJ_category = factor(MAJ_category, levels = levels(DT_category)),
    across(
      c(
        MAJ_task_representation,
        MAJ_task_classification,
        MAJ_task_target,
        MAJ_software
      ),
      function(x) factor(x, levels = c("yes", "no"))
    )
  )

### Majority vs the machine

```{r}
# row-wise majority vote across the three human coders.
# with 3 coders a binary (yes/no) variable always has a majority; for the
# multi-class category a tie (all three disagree) returns NA and is dropped
# from the metrics below.
majority_vote <- function(...) {
  vals <- as.character(c(...))
  vals <- vals[!is.na(vals)]
  if (length(vals) == 0) {
    return(NA_character_)
  }
  tab <- sort(table(vals), decreasing = TRUE)
  if (length(tab) > 1 && tab[[1]] == tab[[2]]) {
    return(NA_character_)
  }
  names(tab)[1]
}

annotation_df_wide_maj <- annotation_df_wide |>
  rowwise() |>
  mutate(
    MAJ_category = majority_vote(DT_category, CV_category, JG_category),
    MAJ_task_representation = majority_vote(
      DT_task_representation,
      CV_task_representation,
      JG_task_representation
    ),
    MAJ_task_classification = majority_vote(
      DT_task_classification,
      CV_task_classification,
      JG_task_classification
    ),
    MAJ_task_target = majority_vote(
      DT_task_target,
      CV_task_target,
      JG_task_target
    ),
    MAJ_software = majority_vote(DT_software, CV_software, JG_software)
  ) |>
  ungroup() |>
  mutate(
    MAJ_category = factor(MAJ_category, levels = levels(DT_category)),
    across(
      c(
        MAJ_task_representation,
        MAJ_task_classification,
        MAJ_task_target,
        MAJ_software
      ),
      function(x) factor(x, levels = c("yes", "no"))
    )
  )
```
